{{- $resource := 0 -}} {{- $url := urls.Parse .Path -}} {{- if not $url.Host | and $url.Path | and (strings.HasSuffix $url.Path "/" | not) -}} {{- $cleanPath := strings.TrimPrefix "/" $url.Path -}} {{- if .Resources -}} {{- with .Resources.GetMatch $url.Path | default (.Resources.GetMatch $cleanPath) -}} {{- $resource = . -}} {{- end -}} {{- end -}} {{- if not $resource -}} {{- with resources.Get $url.Path | default (resources.Get $cleanPath) -}} {{- $resource = . -}} {{- end -}} {{- end -}} {{- end -}} {{- return $resource -}}
